#B.2 Synthetic datasets (generated, not real)

Real lending, M-Pesa, insurance and production data is confidential, so these datasets are invented. They are generated in code with fixed random seeds. They imitate realistic structure, not real people or companies. The cell below lists every file the notebook generated, with its part and size, and the same files are copied to datasets/synthetic/.

import os
from pathlib import Path
import pandas as pd
root = Path(os.environ["GUIDE_ROOT"]) / "work"
rows = []
for p in sorted(root.rglob("*")):
    if p.is_file() and p.suffix.lower() in {".csv", ".xlsx", ".db", ".json", ".md"} and p.stat().st_size > 0:
        rows.append({"part_file": p.parent.name, "file": p.name, "type": p.suffix[1:], "kb": round(p.stat().st_size / 1024, 1)})
cat = pd.DataFrame(rows)
print("generated files:", len(cat), "| total MB:", round(cat.kb.sum() / 1024, 1))
print(cat.groupby("part_file").agg(files=("file", "size"), kb=("kb", "sum")).round(0).to_string())
